from docx import Document
import os


def is_valid_xml_char(char: str) -> bool:
    """
    XML 1.0 valid character ranges.

    python-docx writes Word files as XML internally.
    If a file contains invalid control characters, lxml crashes unless
    we remove those characters first.
    """

    codepoint = ord(char)

    return (
        codepoint == 0x09
        or codepoint == 0x0A
        or codepoint == 0x0D
        or 0x20 <= codepoint <= 0xD7FF
        or 0xE000 <= codepoint <= 0xFFFD
        or 0x10000 <= codepoint <= 0x10FFFF
    )


def clean_text(text: str) -> str:
    """
    Remove every character that is not safe for XML/Word documents.
    """

    return "".join(char for char in text if is_valid_xml_char(char))


def is_binary(file_path):
    """
    Detect whether a file is binary.
    """

    try:
        with open(file_path, "rb") as f:
            chunk = f.read(2048)

        return b"\x00" in chunk

    except Exception:
        return True


def should_skip_file(path, project_path, ignore):
    """
    Decide whether a file should be skipped based on ignored folders/files.
    """

    ignore_folders = set(ignore.get("IGNORE_FOLDERS", []))
    ignore_files = set(ignore.get("IGNORE_FILES", []))

    relative_path = os.path.relpath(path, start=project_path)
    path_parts = relative_path.split(os.sep)
    filename = os.path.basename(path)

    if filename in ignore_files:
        return True

    for part in path_parts:
        if part in ignore_folders:
            return True

    return False


def add_text_safely(doc, relative_path, content):
    """
    Add file content to the Word document safely.

    If a file is very large, split it into chunks so python-docx does not
    try to create one massive XML text node.
    """

    doc.add_heading(relative_path, level=1)

    max_chunk_size = 25000

    if not content.strip():
        doc.add_paragraph("[empty file]")
        return

    for index in range(0, len(content), max_chunk_size):
        chunk = content[index:index + max_chunk_size]
        doc.add_paragraph(chunk)


def build_report(project_path, file_paths, ignore=None, output_path="reports/project-report.docx"):
    """
    Generate a Word document containing readable project source files.

    This version is crash-resistant:
    - skips ignored files
    - skips binary files
    - cleans XML-invalid characters
    - skips only the problematic file if docx insertion fails
    - prints exactly which files were added/skipped
    """

    ignore = ignore or {}

    project_path = os.path.abspath(project_path)
    project_name = os.path.basename(project_path)

    doc = Document()
    doc.add_heading("Project Report", level=0)
    doc.add_paragraph(f"Project: {project_name}")

    added_files = 0
    skipped_ignored = 0
    skipped_binary = 0
    skipped_unreadable = 0
    skipped_docx_error = 0

    for path in file_paths:
        relative_path = os.path.relpath(path, start=project_path)

        if should_skip_file(path, project_path, ignore):
            skipped_ignored += 1
            continue

        if is_binary(path):
            skipped_binary += 1
            print(f"[SKIP binary] {relative_path}")
            continue

        try:
            with open(path, "r", encoding="utf-8", errors="ignore") as f:
                content = f.read()

            content = clean_text(content)

        except Exception as error:
            skipped_unreadable += 1
            print(f"[SKIP unreadable] {relative_path} -> {error}")
            continue

        try:
            add_text_safely(doc, relative_path, content)
            added_files += 1
            print(f"[ADD] {relative_path}")

        except Exception as error:
            skipped_docx_error += 1
            print(f"[SKIP docx-error] {relative_path} -> {error}")
            continue

    os.makedirs(os.path.dirname(output_path), exist_ok=True)
    doc.save(output_path)

    print("\nReport summary:")
    print(f"  Added text files: {added_files}")
    print(f"  Skipped ignored files: {skipped_ignored}")
    print(f"  Skipped binary files: {skipped_binary}")
    print(f"  Skipped unreadable files: {skipped_unreadable}")
    print(f"  Skipped docx-error files: {skipped_docx_error}")
